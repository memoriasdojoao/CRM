Sincronização bidirecional com Google Sheets

Correção Definitiva na Planilha Google

O CRM Santos agora possui deduplicação inteligente e cache local: mesmo se a planilha tiver linhas repetidas ou antigas, ele sempre edita no mesmo registro e exclui de fato sem criar duplicatas.

Para que as linhas sejam editadas e excluídas também fisicamente dentro do Google Sheets, cole o código abaixo no Apps Script da sua planilha:

Código do Google Apps Script (Código.gs) 
function doGet(e) {
  try {
    var sheetName = (e && e.parameter && e.parameter.aba) ? e.parameter.aba : "Levi";
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName(sheetName);
    if (!sheet) {
      sheet = ss.insertSheet(sheetName);
      sheet.appendRow(["Data", "Nome", "Email", "Telefone", "Projeto", "Valor estimado", "Bairro", "Zona", "Status", "Tratativa"]);
      return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON);
    }
    var data = sheet.getDataRange().getValues();
    if (data.length <= 1) return ContentService.createTextOutput(JSON.stringify([])).setMimeType(ContentService.MimeType.JSON);
    var headers = data[0];
    var resultado = [];
    for (var i = 1; i < data.length; i++) {
      var row = data[i];
      var nome = row[1] ? String(row[1]).trim() : "";
      if (!nome) continue;
      var obj = {};
      for (var j = 0; j < headers.length; j++) {
        var val = row[j];
        if (val instanceof Date) val = val.toISOString();
        obj[headers[j]] = (val !== null && val !== undefined) ? String(val) : "";
      }
      resultado.push(obj);
    }
    return ContentService.createTextOutput(JSON.stringify(resultado)).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ error: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  try {
    var postData = (e && e.postData && e.postData.contents) ? e.postData.contents : "{}";
    var dados = JSON.parse(postData);
    var sheetName = dados.aba || "Levi";
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName(sheetName);
    if (!sheet) {
      sheet = ss.insertSheet(sheetName);
      sheet.appendRow(["Data", "Nome", "Email", "Telefone", "Projeto", "Valor estimado", "Bairro", "Zona", "Status", "Tratativa"]);
    }
    var acao = dados.acao || "novo";
    var dataRange = sheet.getDataRange();
    var values = dataRange.getValues();

    if (acao === "excluir") {
      var targetNome = (dados.originalNome || dados.nome || "").trim().toLowerCase();
      if (!targetNome) return ContentService.createTextOutput(JSON.stringify({ status: "erro" })).setMimeType(ContentService.MimeType.JSON);
      for (var i = values.length - 1; i >= 1; i--) {
        var rowNome = String(values[i][1] || "").trim().toLowerCase();
        if (rowNome === targetNome) sheet.deleteRow(i + 1);
      }
      return ContentService.createTextOutput(JSON.stringify({ status: "sucesso", acao: "excluido" })).setMimeType(ContentService.MimeType.JSON);
    }

    if (acao === "editar") {
      var targetNome = (dados.originalNome || "").trim().toLowerCase();
      var editado = false;
      if (targetNome) {
        for (var i = 1; i < values.length; i++) {
          var rowNome = String(values[i][1] || "").trim().toLowerCase();
          if (rowNome === targetNome) {
            sheet.getRange(i + 1, 1, 1, 10).setValues([[
              new Date(), dados.nome || "", dados.email || "", dados.telefone || "",
              dados.projeto || "", dados.valorEstimado || "", dados.bairro || "",
              dados.zona || "", dados.status || "Lead", dados.tratativa || ""
            ]]);
            editado = true;
            break;
          }
        }
      }
      if (!editado) {
        sheet.appendRow([new Date(), dados.nome || "", dados.email || "", dados.telefone || "", dados.projeto || "", dados.valorEstimado || "", dados.bairro || "", dados.zona || "", dados.status || "Lead", dados.tratativa || ""]);
      }
      return ContentService.createTextOutput(JSON.stringify({ status: "sucesso", acao: "editado" })).setMimeType(ContentService.MimeType.JSON);
    }

    if (acao === "novo") {
      sheet.appendRow([new Date(), dados.nome || "", dados.email || "", dados.telefone || "", dados.projeto || "", dados.valorEstimado || "", dados.bairro || "", dados.zona || "", dados.status || "Lead", dados.tratativa || ""]);
      return ContentService.createTextOutput(JSON.stringify({ status: "sucesso", acao: "adicionado" })).setMimeType(ContentService.MimeType.JSON);
    }
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "erro", erro: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}
